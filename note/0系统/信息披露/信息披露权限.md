@SuppressWarnings("unchecked")
	public List<Object> queryRoelTemplate(String isRoot, String queryParam) {
		getXbrlReportService();
		DBType dbtype = xbrlReportService.getDBType();
	    DBUtil.setDbType(dbtype);
    	Dao dao = this.getDao(getDataSourceMapping());
        StringBuilder sql = new StringBuilder();
        sql.append("SELECT D.FPARENT_ID,concat(D.FREPORT_NAME_FIR , D.FREPORT_NAME_SEC) as FREPORT_NAME,D.FREPORT_TYPE FROM (SELECT -1 AS FPARENT_ID, CASE WHEN (a.FPERIOD = 'SEASON' OR a.FPERIOD = 'quarter') THEN '季报' WHEN a.FPERIOD = 'YEAR' THEN '年报' ");
        sql.append("WHEN a.FPERIOD = 'HALFYEAR' THEN '中期报告' WHEN a.FPERIOD = 'DAY' THEN '日报' WHEN a.FPERIOD = 'MONTH' THEN '月报' ELSE '' END AS freport_name_sec, ");
        sql.append("CASE WHEN a.FREPORT_TYPE_SIN LIKE '%GM%' AND a.FPERIOD <> 'TEMPLATE' AND a.FPERIOD <> 'ZZPL' THEN '公募基金' WHEN a.FREPORT_TYPE_SIN LIKE '%GM%' AND a.FPERIOD = 'TEMPLATE' THEN '招募说明书' WHEN a.FREPORT_TYPE_SIN LIKE '%GM%' AND a.FPERIOD = 'ZZPL' THEN '自主披露报告' ");
        sql.append("when a.FREPORT_TYPE_SIN = 'XB_BBLX_CONFIG' then '可配置化定期报告' ");
        sql.append("WHEN a.FREPORT_TYPE_SIN LIKE '%LCCP%' THEN '理财产品' WHEN a.FREPORT_TYPE_SIN LIKE '%NJ%' THEN '年金' ");
        sql.append("WHEN a.FREPORT_TYPE_SIN LIKE '%QD%' THEN 'QDII' WHEN a.FREPORT_TYPE_SIN LIKE '%RMBLC%' THEN '人民币理财' WHEN a.FREPORT_TYPE_SIN LIKE '%SM%' THEN '私募基金' ");
        sql.append("WHEN a.FREPORT_TYPE_SIN LIKE '%TGBG%' AND a.FPERIOD <> ' ' THEN '托管报告' WHEN a.FREPORT_TYPE_SIN LIKE '%TGBG%' AND a.FPERIOD = ' ' THEN '非标准化报告' ");
        sql.append("WHEN a.FREPORT_TYPE_SIN LIKE '%LZBG%' AND a. FPERIOD <> ' ' THEN '托管人履职报告' ");
        sql.append("WHEN a.FREPORT_TYPE_SIN LIKE '%TEMPLATE%' THEN '临时公告' WHEN a.FREPORT_TYPE_SIN LIKE '%OTHER%' THEN '其他' when a.FREPORT_TYPE_SIN like '%DZHDQ%' then '定制化定期报告' when a.FREPORT_TYPE_SIN like '%FHH%' then '复核函' ELSE null END AS freport_name_fir, a.FPERIOD,a.FREPORT_TYPE,a.FREPORT_CLASS_TYPE FROM (");
        sql.append("SELECT t.FREPORT_TYPE AS FREPORT_TYPE_SIN, t.FPERIOD,"+DBUtil.sqlJN("t.FREPORT_TYPE","':'", "t.FPERIOD"));
        sql.append(" AS FREPORT_TYPE , '' AS FREPORT_CLASS_TYPE FROM T_XB_TEMPLATE_INFO t GROUP BY t.FREPORT_TYPE,t.FPERIOD) a ) D WHERE D.freport_name_fir IS NOT NULL   ORDER BY concat(D.FREPORT_NAME_FIR , D.FREPORT_NAME_SEC)--SQLTD");
        return (List<Object>) dao.find(sql.toString(), new SimpleResultSetHandler());
	}
